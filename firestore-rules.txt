rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {
    
    // Only the owning resort may read its guest uploads. Mutations use server APIs.
    match /guestbooking/{bookingId} {
      allow create: if false;
      allow read: if request.auth != null
        && resource.data.get("ownerUid", "") == request.auth.uid;
      allow update, delete: if false;
    }

    // OTP challenges and date locks are server-only records.
    match /bookingOtpChallenges/{challengeId} {
      allow read, write: if false;
    }
    match /bookingDateLocks/{lockId} {
      allow read, write: if false;
    }
    match /paymentHistory/{paymentId} {
      allow read, write: if false;
    }

    // Owner documents contain private account data and payment settings.
    // Only the owner and authorized super admins may read them.
    match /resortOwners/{ownerId} {
      function isSuperAdminEmail() {
        return request.auth != null
          && request.auth.token.email != null
          && request.auth.token.email_verified == true
          && (request.auth.token.email == "ecommerceresortadmin@gmail.com"
              || request.auth.token.email == "commerceresortadmin@gmail.com");
      }

      allow read: if request.auth != null
        && (request.auth.uid == ownerId || isSuperAdminEmail());
      allow create: if request.auth != null
        && request.auth.uid == ownerId
        && request.resource.data.uid == ownerId
        && request.resource.data.get("status", "pending") == "pending";
      allow update: if isSuperAdminEmail()
        || (request.auth != null
          && request.auth.uid == ownerId
          && request.resource.data.uid == resource.data.uid
          && request.resource.data.role == resource.data.role
          && request.resource.data.status == resource.data.status
          && request.resource.data.diff(resource.data).affectedKeys().hasOnly([
            "email",
            "displayName",
            "photoURL",
            "createdAt",
            "updatedAt",
            "gcashQrImage",
            "paymentInstructions",
            "checkInInstructions",
            "resortRulesText"
          ]));
      allow delete: if isSuperAdminEmail();

      match /staffInvitations/{email} {
        allow read: if request.auth != null
          && (request.auth.uid == ownerId
            || (request.auth.token.email == email
              && resource.data.get("active", false) == true));
        allow write: if false;
      }

      match /staffNotifications/{notificationId} {
        allow read: if request.auth != null
          && resource.data.get("staffUid", "") == request.auth.uid;
        allow update: if request.auth != null
          && resource.data.get("staffUid", "") == request.auth.uid
          && request.resource.data.get("staffUid", "") == request.auth.uid
          && request.resource.data.get("read", false) == true
          && request.resource.data.diff(resource.data).affectedKeys().hasOnly(["read", "readAt"]);
        allow create, delete: if false;
      }
    }

    // Resort listings are public. Owners may submit their own listings for review;
    // only a verified Super Admin may approve, edit, create, or delete listings.
    match /resorts/{resortId} {
      function isSuperAdminEmail() {
        return request.auth != null
          && request.auth.token.email != null
          && request.auth.token.email_verified == true
          && (request.auth.token.email == "ecommerceresortadmin@gmail.com"
              || request.auth.token.email == "commerceresortadmin@gmail.com");
      }

      allow read: if true;
      allow create: if isSuperAdminEmail()
        || (request.auth != null
          && resortId == request.auth.uid
          && request.resource.data.ownerUid == request.auth.uid
          && request.resource.data.get("status", "pending") == "pending");
      allow update: if isSuperAdminEmail()
        || (request.auth != null
          && resource.data.get("ownerUid", "") == request.auth.uid
          && request.resource.data.ownerUid == request.auth.uid
          && request.resource.data.status == "pending"
          && request.resource.data.diff(resource.data).affectedKeys().hasOnly([
            "ownerEmail",
            "name",
            "location",
            "mapsUrl",
            "description",
            "status",
            "updatedAt",
            "createdAt"
          ]));
      allow delete: if isSuperAdminEmail();
    }
    
    // Rooms are public, but writes belong to the authenticated room owner.
    // Internal helpdesk accounts may claim legacy room docs with no ownerUid.
    match /rooms/{roomId} {
      function isLegacyRoomsAdmin() {
        return request.auth != null
          && request.auth.token.email != null
          && request.auth.token.email_verified == true
          && (request.auth.token.email == "admin@luxestay.com"
              || request.auth.token.email == "resort.helpdesk01@gmail.com");
      }

      function isUnscopedRoom() {
        return resource.data.get("ownerUid", null) == null
          || resource.data.get("ownerUid", "") == "";
      }

      allow read: if true;
      allow create: if request.auth != null
        && request.resource.data.ownerUid == request.auth.uid;
      allow update: if request.auth != null
        && (
          (resource.data.get("ownerUid", "") == request.auth.uid
            && request.resource.data.ownerUid == request.auth.uid)
          || (isLegacyRoomsAdmin()
            && isUnscopedRoom()
            && request.resource.data.ownerUid == request.auth.uid)
        );
      allow delete: if request.auth != null
        && (
          resource.data.get("ownerUid", "") == request.auth.uid
          || (isLegacyRoomsAdmin() && isUnscopedRoom())
        );
    }
    
    // Contact messages collection
    match /contactMessages/{document=**} {
      allow create: if true;  // Anyone can send contact messages
      allow read, update, delete: if request.auth != null;  // Only admins can view/update/delete
    }
    
    // Feedbacks collection
    match /feedbacks/{document=**} {
      allow create: if true;  // Anyone can submit feedback
      allow read: if true;  // Public can view published feedbacks
      allow update, delete: if request.auth != null;  // Only admins can update/delete
    }
    
    // Per-owner outbound email (Gmail) — only that signed-in owner can read/write.
    match /resortOwners/{ownerId}/secrets/{docId} {
      allow read, write: if request.auth != null && request.auth.uid == ownerId;
    }

    // Per-resort public site branding (landing + admin). Anyone can read; only that owner can write.
    match /resortOwners/{ownerId}/site/{docId} {
      allow read: if true;
      allow write: if request.auth != null && request.auth.uid == ownerId;
    }

    // Super Admin audit log — read/create only for authorized emails (append-only).
    match /superAdminAuditLog/{logId} {
      function isSuperAdminEmail() {
        return request.auth != null
          && request.auth.token.email != null
          && request.auth.token.email_verified == true
          && (request.auth.token.email == "ecommerceresortadmin@gmail.com"
              || request.auth.token.email == "commerceresortadmin@gmail.com");
      }
      allow read: if isSuperAdminEmail();
      allow create: if isSuperAdminEmail();
      allow update, delete: if false;
    }

    match /superAdminNotifications/{notificationId} {
      function isVerifiedSuperAdminEmail() {
        return request.auth != null
          && request.auth.token.email != null
          && request.auth.token.email_verified == true
          && (request.auth.token.email == "ecommerceresortadmin@gmail.com"
              || request.auth.token.email == "commerceresortadmin@gmail.com");
      }

      allow read: if isVerifiedSuperAdminEmail();
      allow create: if isVerifiedSuperAdminEmail()
        && request.resource.data.get("actorUid", "") == request.auth.uid
        && request.resource.data.get("read", false) == false
        && request.resource.data.keys().hasOnly([
          "type",
          "title",
          "message",
          "resortId",
          "ownerUid",
          "actorUid",
          "read",
          "createdAt"
        ]);
      allow update: if isVerifiedSuperAdminEmail()
        && request.resource.data.get("read", false) == true
        && request.resource.data.diff(resource.data).affectedKeys().hasOnly(["read", "readAt"]);
      allow delete: if false;
    }

    match /resortNotificationOutbox/{eventId} {
      function isVerifiedSuperAdminEmail() {
        return request.auth != null
          && request.auth.token.email != null
          && request.auth.token.email_verified == true
          && (request.auth.token.email == "ecommerceresortadmin@gmail.com"
              || request.auth.token.email == "commerceresortadmin@gmail.com");
      }

      allow create: if isVerifiedSuperAdminEmail()
        && request.resource.data.get("eventId", "") == eventId
        && request.resource.data.get("actorUid", "") == request.auth.uid
        && request.resource.data.get("entityType", "") == "platform"
        && request.resource.data.get("platformAdminRecipient", false) == true
        && request.resource.data.get("status", "") == "pending"
        && request.resource.data.get("attempts", -1) == 0
        && request.resource.data.keys().hasOnly([
          "ownerUid",
          "eventId",
          "action",
          "entityType",
          "entityId",
          "audienceRoles",
          "title",
          "message",
          "actorUid",
          "platformAdminRecipient",
          "status",
          "attempts",
          "createdAt"
        ]);
      allow read, update, delete: if false;
    }

    // Settings: public read except resort SMTP doc (super admin only).
    match /settings/{document} {
      function isSuperAdminEmail() {
        return request.auth != null
          && request.auth.token.email != null
          && (request.auth.token.email == "ecommerceresortadmin@gmail.com"
              || request.auth.token.email == "commerceresortadmin@gmail.com");
      }
      allow read: if document != "resortNotifySmtp" || isSuperAdminEmail();
      // Marketplace copy: only super admins may change (any signed-in user can still edit branding, etc.).
      allow write: if document == "resortNotifySmtp"
        ? isSuperAdminEmail()
        : document == "marketplace"
          ? isSuperAdminEmail()
          : request.auth != null;
    }
    
    // Unlisted collections require authentication. Exclude collections with
    // explicit rules so this fallback cannot bypass them (Firestore rules are ORed).
    match /{collection}/{document=**} {
      allow read, write: if !(collection in [
        "rooms",
        "guestbooking",
        "bookingOtpChallenges",
        "bookingDateLocks",
        "paymentHistory",
        "resortOwners",
        "resorts",
        "superAdminAuditLog",
        "superAdminNotifications",
        "resortNotificationOutbox"
      ]) && request.auth != null;
    }
  }
}
